cmake_minimum_required(VERSION 3.20)
project(EDF6VehicleImpact VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(EDF6VehicleImpact SHARED src/plugin.cpp src/memory.cpp)
target_include_directories(EDF6VehicleImpact PRIVATE third_party/EDFModLoader)
target_compile_options(EDF6VehicleImpact PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_options(EDF6VehicleImpact PRIVATE /DYNAMICBASE /NXCOMPAT /PDBALTPATH:EDF6VehicleImpact.pdb)
set_target_properties(EDF6VehicleImpact PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist")
configure_file(EDF6VehicleImpact.ini "${CMAKE_SOURCE_DIR}/dist/EDF6VehicleImpact.ini" COPYONLY)
